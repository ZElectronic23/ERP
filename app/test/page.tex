import { supabase } from '@/lib/supabaseClient'

export default async function TestPage() {
  // نحاول نجيب المنتجات من جدول products
  const { data: products, error } = await supabase
    .from('products')
    .select('*')
    .limit(5)

  // نحاول نجيب عدد المنتجات فقط كاختبار تاني
  const { count, error: countError } = await supabase
    .from('products')
    .select('*', { count: 'exact', head: true })

  return (
    <div dir="rtl" className="p-8 max-w-4xl mx-auto">
      <h1 className="text-3xl font-bold mb-6 text-center">🔍 اختبار الاتصال بقاعدة البيانات</h1>
      
      {/* معلومات الاتصال */}
      <div className="mb-6 p-4 border rounded-lg bg-blue-50">
        <h2 className="font-bold text-lg mb-2">📌 معلومات الاتصال:</h2>
        <p><span className="font-semibold">الجدول:</span> products</p>
        <p><span className="font-semibold">URL:</span> {process.env.NEXT_PUBLIC_SUPABASE_URL}</p>
        <p><span className="font-semibold">الحالة:</span> 
          {process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY ? 
            ' ✅ المفتاح موجود' : 
            ' ❌ المفتاح ناقص'}
        </p>
      </div>

      {/* عرض الأخطاء إن وجدت */}
      {error && (
        <div className="bg-red-50 text-red-700 p-4 rounded-lg mb-6 border border-red-200">
          <h2 className="font-bold text-lg mb-2">❌ خطأ من Supabase:</h2>
          <p className="font-mono text-sm">{JSON.stringify(error, null, 2)}</p>
        </div>
      )}

      {/* عرض عدد المنتجات */}
      <div className="mb-6 p-4 border rounded-lg bg-green-50">
        <h2 className="font-bold text-lg mb-2">📊 معلومات المنتجات:</h2>
        <p><span className="font-semibold">عدد المنتجات المسترجعة:</span> {products?.length || 0}</p>
        <p><span className="font-semibold">إجمالي عدد المنتجات في الجدول:</span> {count || 0}</p>
      </div>

      {/* عرض أول منتج إن وجد */}
      {products && products.length > 0 ? (
        <div className="mb-6">
          <h2 className="font-bold text-lg mb-2">📋 أول منتج في الجدول:</h2>
          <div className="bg-gray-100 p-4 rounded-lg">
            <pre className="text-sm overflow-auto">
              {JSON.stringify(products[0], null, 2)}
            </pre>
          </div>
        </div>
      ) : (
        <div className="bg-yellow-50 p-4 rounded-lg mb-6 border border-yellow-200">
          <p className="text-yellow-700">⚠️ لا توجد بيانات في جدول products</p>
          <p className="text-sm mt-2">تأكد من:</p>
          <ul className="list-disc list-inside text-sm mt-1">
            <li>إن الجدول فيه بيانات (روح لـ Table Editor في Supabase)</li>
            <li>إن اسم الجدول بالظبط products (مش Products أو PRODUCTS)</li>
            <li>إن الـ API keys صح ولها صلاحية قراءة الجدول</li>
          </ul>
        </div>
      )}
    </div>
  )
}
